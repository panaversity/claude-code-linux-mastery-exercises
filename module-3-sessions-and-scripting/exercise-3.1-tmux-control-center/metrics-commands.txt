# System Metrics Commands for Monitoring Window
# Run these to monitor server health while agents operate

# CPU and memory overview (updates every 2 seconds)
watch -n 2 'echo "=== CPU ===" && uptime && echo "" && echo "=== MEMORY ===" && free -h && echo "" && echo "=== DISK ===" && df -h / /var /opt'
